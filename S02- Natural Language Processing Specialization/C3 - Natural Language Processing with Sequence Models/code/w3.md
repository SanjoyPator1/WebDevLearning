## Part 1: Setup and Introduction

### 1.1 What are Siamese Networks?

A **Siamese Network** uses the same weights while processing two different inputs to produce comparable output vectors. The key insight: similar inputs should produce nearby vectors in the embedding space.

**Architecture:**

```
Sentence 1 → [Shared Encoder] → Vector v1 ↘
                                              [Similarity] → Score
Sentence 2 → [Shared Encoder] → Vector v2 ↗

Note: Both encoders share the SAME weights!
```

**Applications:**

- Question duplicate detection
- Semantic search
- Plagiarism detection
- Face verification
- Product matching

class TripletLoss(keras.losses.Loss):
    """
    Triplet loss with hard negative mining

    Args:
        margin (float): Margin for triplet loss
    """

    def __init__(self, margin=0.25, name='triplet_loss'):
        super().__init__(name=name)
        self.margin = margin

    def call(self, y_true, y_pred):
        """
        Compute triplet loss

        Args:
            y_true: Dummy labels (not used, but required by Keras)
            y_pred: Model output [v1, v2] concatenated

        Returns:
            loss: Triplet loss value
        """
        ### START CODE HERE ###

        # Split y_pred into v1 and v2
        embedding_size = tf.shape(y_pred)[1] // 2
        v1 = None  # Your code: y_pred[:, :embedding_size]
        v2 = None  # Your code: y_pred[:, embedding_size:]

        # Compute similarity matrix: S = v2 @ v1^T
        # Each element S[i,j] is similarity between v2[i] and v1[j]
        # Hint: Use tf.linalg.matmul with transpose_b=True
        sim_matrix = None  # Your code

        # Get batch size
        batch_size = tf.cast(tf.shape(v1)[0], tf.float32)

        # Positive similarities (diagonal of similarity matrix)
        # These are similarities between matching pairs
        # Hint: Use tf.linalg.diag_part
        positive_sims = None  # Your code

        # Create mask to exclude positives from negatives
        # Hint: Use tf.eye to create identity matrix
        mask_positives = tf.eye(tf.shape(sim_matrix)[0])

        # Zero out positive similarities in the matrix
        # Hint: Subtract diagonal using tf.linalg.diag
        negative_sims = None  # Your code: sim_matrix - tf.linalg.diag(positive_sims)

        # Compute mean negative (average of all negatives per anchor)
        # Hint: Sum over axis=1 and divide by (batch_size - 1)
        mean_negative = None  # Your code

        # Compute closest negative (hardest negative per anchor)
        # Hint: Create a mask to exclude positives, then find max
        mask_exclude = tf.cast(tf.equal(mask_positives, 0), tf.float32)
        negative_sims_masked = negative_sims + (1 - mask_exclude) * (-1e9)  # Large negative for excluded
        closest_negative = tf.reduce_max(negative_sims_masked, axis=1)

        # Loss 1: Mean negative loss
        # Hint: max(0, -positive + mean_negative + margin)
        loss1 = tf.maximum(0.0, -positive_sims + mean_negative + self.margin)

        # Loss 2: Closest negative loss
        # Hint: max(0, -positive + closest_negative + margin)
        loss2 = tf.maximum(0.0, -positive_sims + closest_negative + self.margin)

        # Total loss (average of both losses)
        total_loss = tf.reduce_mean(loss1 + loss2)

        ### END CODE HERE ###

        return total_loss

### 5.2 Alternative: Contrastive Loss

```python
class ContrastiveLoss(keras.losses.Loss):
    """
    Contrastive loss for Siamese networks
    Alternative to triplet loss

    Args:
        margin (float): Margin for dissimilar pairs
    """

    def __init__(self, margin=1.0, name='contrastive_loss'):
        super().__init__(name=name)
        self.margin = margin

    def call(self, y_true, y_pred):
        """
        Compute contrastive loss

        Args:
            y_true: True labels (1 for similar, 0 for dissimilar)
            y_pred: Model output [v1, v2]

        Returns:
            loss: Contrastive loss
        """
        # Split into v1 and v2
        embedding_size = tf.shape(y_pred)[1] // 2
        v1 = y_pred[:, :embedding_size]
        v2 = y_pred[:, embedding_size:]

        # Compute cosine similarity (since vectors are normalized)
        similarity = tf.reduce_sum(v1 * v2, axis=1)

        # Contrastive loss
        # For similar pairs: penalize low similarity
        # For dissimilar pairs: penalize high similarity (above margin)
        loss_similar = y_true * (1 - similarity)
        loss_dissimilar = (1 - y_true) * tf.maximum(0.0, similarity - self.margin)

        loss = tf.reduce_mean(loss_similar + loss_dissimilar)

        return loss

class SimilarityAccuracy(keras.metrics.Metric):
    """
    Custom accuracy metric for Siamese networks
    """

    def __init__(self, threshold=0.5, name='similarity_accuracy', **kwargs):
        super().__init__(name=name, **kwargs)
        self.threshold = threshold
        self.correct = self.add_weight(name='correct', initializer='zeros')
        self.total = self.add_weight(name='total', initializer='zeros')

    def update_state(self, y_true, y_pred, sample_weight=None):
        # Split y_pred
        embedding_size = tf.shape(y_pred)[1] // 2
        v1 = y_pred[:, :embedding_size]
        v2 = y_pred[:, embedding_size:]

        # Compute similarity
        similarity = tf.reduce_sum(v1 * v2, axis=1)

        # Predictions
        predictions = tf.cast(similarity > self.threshold, tf.float32)

        # Check matches
        matches = tf.cast(tf.equal(predictions, y_true), tf.float32)

        self.correct.assign_add(tf.reduce_sum(matches))
        self.total.assign_add(tf.cast(tf.shape(y_true)[0], tf.float32))

    def result(self):
        return self.correct / self.total

    def reset_state(self):
        self.correct.assign(0.0)
        self.total.assign(0.0)
```

def check_similarity(sentence1, sentence2, model, threshold=0.5):
    """
    Check if two sentences are similar

    Args:
        sentence1 (str): First sentence
        sentence2 (str): Second sentence
        model: Trained Siamese model
        threshold (float): Similarity threshold

    Returns:
        dict: Results with similarity score and prediction
    """
    # Prepare input
    input_data = ([sentence1], [sentence2])

    # Get prediction
    output = model.predict(input_data, verbose=0)

    # Split and compute similarity
    embedding_size = output.shape[1] // 2
    v1 = output[:, :embedding_size]
    v2 = output[:, embedding_size:]

    similarity = np.sum(v1 * v2, axis=1)[0]
    is_similar = similarity > threshold

    return {
        'similarity': similarity,
        'is_similar': is_similar,
        'threshold': threshold
    }

def find_most_similar(query, sentences, model, top_k=5):
    """
    Find most similar sentences to a query

    Args:
        query (str): Query sentence
        sentences (list): List of sentences to search
        model: Siamese model
        top_k (int): Number of results to return

    Returns:
        list: Top-k most similar sentences with scores
    """
    # Create input pairs
    query_repeated = [query] * len(sentences)

    # Get predictions
    predictions = model.predict(
        (np.array(query_repeated), np.array(sentences)),
        batch_size=128,
        verbose=0
    )

    # Compute similarities
    embedding_size = predictions.shape[1] // 2
    v1 = predictions[:, :embedding_size]
    v2 = predictions[:, embedding_size:]
    similarities = np.sum(v1 * v2, axis=1)

    # Get top-k
    top_indices = np.argsort(similarities)[::-1][:top_k]

    results = [
        (sentences[idx], similarities[idx])
        for idx in top_indices
    ]

    return results
