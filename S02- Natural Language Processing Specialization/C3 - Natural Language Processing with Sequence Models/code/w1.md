def generate_text_greedy(model, start_string, num_chars, char_to_idx, idx_to_char):
    """
    Generate text using greedy sampling (always pick most likely character)

    Args:
        model: Trained model
        start_string: Starting text
        num_chars: Number of characters to generate
        char_to_idx: Character to index mapping
        idx_to_char: Index to character mapping

    Returns:
        str: Generated text
    """
    # Convert start string to tensor
    input_tensor = text_to_tensor(start_string, char_to_idx)
    input_tensor = tf.expand_dims(input_tensor, 0)  # Add batch dimension

    # Initialize
    generated_text = start_string
    states = None

    # Generate characters one by one
    for i in range(num_chars):
        # Get predictions
        predictions, states = model(input_tensor, states=states, return_state=True)

        # Get last character prediction
        predictions = predictions[:, -1, :]

        # Greedy: pick character with highest probability
        predicted_id = tf.argmax(predictions, axis=-1).numpy()[0]

        # Convert to character
        predicted_char = idx_to_char[predicted_id]
        generated_text += predicted_char

        # Update input for next iteration
        input_tensor = tf.expand_dims([predicted_id], 0)

    return generated_text

def generate_text_temperature(model, start_string, num_chars,
                              char_to_idx, idx_to_char, temperature=1.0):
    """
    Generate text using temperature sampling

    Args:
        model: Trained model
        start_string: Starting text
        num_chars: Number of characters to generate
        char_to_idx: Character to index mapping
        idx_to_char: Index to character mapping
        temperature: Sampling temperature (higher = more random)

    Returns:
        str: Generated text
    """
    # Convert start string to tensor
    input_tensor = text_to_tensor(start_string, char_to_idx)
    input_tensor = tf.expand_dims(input_tensor, 0)

    # Initialize
    generated_text = start_string
    states = None

    # Generate characters
    for i in range(num_chars):
        # Get predictions
        predictions, states = model(input_tensor, states=states, return_state=True)

        # Get last character prediction
        predictions = predictions[:, -1, :]

        # Apply temperature
        predictions = predictions / temperature

        # Sample from distribution
        predicted_id = tf.random.categorical(predictions, num_samples=1)[-1, 0].numpy()

        # Convert to character
        predicted_char = idx_to_char[predicted_id]
        generated_text += predicted_char

        # Update input
        input_tensor = tf.expand_dims([predicted_id], 0)

    return generated_text
