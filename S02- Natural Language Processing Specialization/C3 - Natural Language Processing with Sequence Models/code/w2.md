## Part 1: Setup and Introduction

### 1.1 What is Named Entity Recognition?

Named Entity Recognition (NER) is the task of identifying and classifying named entities in text into predefined categories.

**Example:**

```
Input:  "Apple Inc. was founded by Steve Jobs in California."
Output: Apple Inc. [ORG]
        Steve Jobs [PER]
        California [LOC]
```

**Common Entity Types:**

- **PER** (Person): Steve Jobs, Albert Einstein
- **ORG** (Organization): Apple Inc., UN
- **LOC** (Location): California, Paris
- **MISC** (Miscellaneous): Other entities

**BIO Tagging Scheme:**

- **B-**: Beginning of an entity
- **I-**: Inside/continuation of an entity
- **O**: Outside any entity (not an entity)

Example:

```
Steve  B-PER
Jobs   I-PER
founded O
Apple  B-ORG
Inc    I-ORG
.      O
```

    """
    Convert sentences to padded index sequences

    Args:
        sentences (list): List of sentences (each is list of words)

    Returns:
        sequences (np.array): Padded sequences of indices
    """

    # Determine max length if not provided
    if max_len is None:
        max_len = max(len(sent) for sent in sentences)

    # Convert sentences to indices
    sequences = []
    for sentence in sentences:
        # Convert each word to index (use 1 for <UNK> if word not in vocab)

class BiLSTM_NER(keras.Model):
    """
    Bidirectional LSTM for Named Entity Recognition

    Args:
        vocab_size (int): Size of vocabulary
        num_labels (int): Number of NER labels
        embedding_dim (int): Dimension of word embeddings
        lstm_units (int): Number of LSTM units
        dropout_rate (float): Dropout rate
    """

    def __init__(self, vocab_size, num_labels, embedding_dim=128,
                 lstm_units=256, dropout_rate=0.3):
        super(BiLSTM_NER, self).__init__()

        self.vocab_size = vocab_size
        self.num_labels = num_labels
        self.embedding_dim = embedding_dim
        self.lstm_units = lstm_units

        ### START CODE HERE ###

        # 1. Embedding layer
        # Hint: layers.Embedding(input_dim=vocab_size, output_dim=embedding_dim, mask_zero=True)
        # mask_zero=True tells the model to ignore padded values (index 0)
        self.embedding = None

        # 2. Dropout after embedding
        self.embedding_dropout = layers.Dropout(dropout_rate)

        # 3. Bidirectional LSTM layer
        # Hint: layers.Bidirectional(layers.LSTM(units=lstm_units, return_sequences=True))
        # return_sequences=True because we need output for each token
        self.bilstm = None

        # 4. Dropout after LSTM
        self.lstm_dropout = layers.Dropout(dropout_rate)

        # 5. Dense output layer
        # Hint: layers.Dense(units=num_labels)
        # No activation here - we'll use it in the loss function
        self.dense = None

        ### END CODE HERE ###

    def call(self, inputs, training=False):
        """
        Forward pass

        Args:
            inputs: Input tensor of shape (batch_size, seq_length)
            training: Whether in training mode

        Returns:
            Output tensor of shape (batch_size, seq_length, num_labels)
        """
        # Embedding
        x = self.embedding(inputs)
        x = self.embedding_dropout(x, training=training)

        # BiLSTM
        x = self.bilstm(x)
        x = self.lstm_dropout(x, training=training)

        # Dense output
        output = self.dense(x)

        return output

class MaskedSparseCategoricalCrossentropy(keras.losses.Loss):
    """
    Masked Sparse Categorical Crossentropy Loss
    Ignores padded positions (label_id = 0)
    """

    def __init__(self, name='masked_sparse_categorical_crossentropy'):
        super().__init__(name=name)

    def call(self, y_true, y_pred):
        """
        Compute masked loss

        Args:
            y_true: True labels (batch_size, seq_length)
            y_pred: Predicted logits (batch_size, seq_length, num_labels)

        Returns:
            loss: Scalar loss value
        """
        ### START CODE HERE ###

        # Create mask (1 for real tokens, 0 for padding)
        # Hint: Padding has label_id = 0
        mask = tf.cast(tf.not_equal(None, 0), dtype=tf.float32)  # Your code: y_true

        # Compute sparse categorical crossentropy
        # Hint: Use keras.losses.sparse_categorical_crossentropy
        # Set from_logits=True since we don't have softmax in the model
        loss = keras.losses.sparse_categorical_crossentropy(
            None,  # Your code: y_true
            None,  # Your code: y_pred
            from_logits=True
        )

        # Apply mask
        loss = loss * mask

        # Compute average loss (only over non-padded positions)
        loss = tf.reduce_sum(loss) / tf.reduce_sum(mask)

        ### END CODE HERE ###

        return loss

class MaskedAccuracy(keras.metrics.Metric):
    """
    Masked Accuracy Metric
    Only computes accuracy on non-padded positions
    """

    def __init__(self, name='masked_accuracy', **kwargs):
        super().__init__(name=name, **kwargs)
        self.total = self.add_weight(name='total', initializer='zeros')
        self.count = self.add_weight(name='count', initializer='zeros')

    def update_state(self, y_true, y_pred, sample_weight=None):
        """
        Update metric state

        Args:
            y_true: True labels (batch_size, seq_length)
            y_pred: Predicted logits (batch_size, seq_length, num_labels)
        """
        ### START CODE HERE ###

        # Get predicted classes
        # Hint: Use tf.argmax on the last axis
        y_pred_classes = tf.argmax(None, axis=-1)  # Your code: y_pred

        # Cast to same type
        y_true = tf.cast(y_true, tf.int64)
        y_pred_classes = tf.cast(y_pred_classes, tf.int64)

        # Create mask
        mask = tf.cast(tf.not_equal(y_true, 0), dtype=tf.float32)

        # Check matches
        matches = tf.cast(tf.equal(y_true, y_pred_classes), dtype=tf.float32)

        # Apply mask
        matches = matches * mask

        # Update totals
        self.total.assign_add(tf.reduce_sum(matches))
        self.count.assign_add(tf.reduce_sum(mask))

        ### END CODE HERE ###

    def result(self):
        """Return the metric result"""
        return self.total / self.count

    def reset_state(self):
        """Reset metric state"""
        self.total.assign(0.0)
        self.count.assign(0.0)

def extract_entities(sentence, labels, idx2label):
    """
    Extract entities from a sentence with BIO labels

    Args:
        sentence (list): List of words
        labels (list): List of label indices
        idx2label (dict): Index to label mapping

    Returns:
        entities (list): List of (entity_text, entity_type) tuples
    """
    ### START CODE HERE ###

    entities = []
    current_entity = []
    current_type = None

    for word, label_idx in zip(sentence, labels):
        label = idx2label[label_idx]

        if label.startswith('B-'):
            # Beginning of new entity
            if current_entity:
                # Save previous entity
                entities.append((' '.join(current_entity), current_type))

            # Start new entity
            current_entity = [word]
            current_type = label[2:]  # Remove 'B-' prefix

        elif label.startswith('I-') and current_entity:
            # Inside entity (continuation)
            entity_type = label[2:]  # Remove 'I-' prefix
            if entity_type == current_type:
                current_entity.append(word)

        else:
            # Outside entity ('O') or other
            if current_entity:
                entities.append((' '.join(current_entity), current_type))
                current_entity = []
                current_type = None

    # Don't forget last entity
    if current_entity:
        entities.append((' '.join(current_entity), current_type))

    ### END CODE HERE ###

    return entities

def predict_sentence(sentence, model, word2idx, idx2label, max_len):
    """
    Predict NER labels for a sentence

    Args:
        sentence (list): List of words
        model: Trained model
        word2idx (dict): Word to index mapping
        idx2label (dict): Index to label mapping
        max_len (int): Maximum sequence length

    Returns:
        predictions (list): List of predicted labels
    """
    # Convert to indices
    indices = [word2idx.get(word, 1) for word in sentence]  # 1 is <UNK>

    # Pad
    if len(indices) < max_len:
        indices = indices + [0] * (max_len - len(indices))
    else:
        indices = indices[:max_len]

    # Predict
    input_tensor = tf.constant([indices])
    predictions = model.predict(input_tensor, verbose=0)

    # Get predicted labels
    predicted_indices = tf.argmax(predictions[0], axis=-1).numpy()

    # Convert back to labels (only for actual sentence length)
    predicted_labels = [idx2label[idx] for idx in predicted_indices[:len(sentence)]]

    return predicted_labels
