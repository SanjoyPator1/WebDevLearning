```python
def sigmoid(Z):
    """
    Sigmoid activation function with numerical stability

    Formula: σ(z) = 1 / (1 + e^(-z))
    """
    # Clip Z to prevent overflow
    Z = np.clip(Z, -500, 500)
    return 1 / (1 + np.exp(-Z))

def relu(Z):
    """
    ReLU activation function

    Formula: ReLU(z) = max(0, z)
    """
    return np.maximum(0, Z)

def tanh_activation(Z):
    """
    Tanh activation function

    Formula: tanh(z) = (e^z - e^(-z)) / (e^z + e^(-z))
    """
    return np.tanh(Z)
```

## Cell 5: Code - Activation Function Derivatives

```python
def sigmoid_backward(dA, cache):
    """
    Backward propagation for sigmoid activation

    Formula: g'(z) = g(z)(1 - g(z))
    """
    Z = cache
    s = sigmoid(Z)
    dZ = dA * s * (1 - s)
    return dZ

def relu_backward(dA, cache):
    """
    Backward propagation for ReLU activation

    Formula: g'(z) = 1 if z > 0, else 0
    """
    Z = cache
    dZ = np.array(dA, copy=True)
    dZ[Z <= 0] = 0  # Zero gradient when input was negative
    return dZ

def tanh_backward(dA, cache):
    """
    Backward propagation for tanh activation

    Formula: g'(z) = 1 - tanh²(z)
    """
    Z = cache
    tanh_z = np.tanh(Z)
    dZ = dA * (1 - tanh_z**2)
    return dZ
```

```markdown
### General Dimension Rules
```

## Cell 7: Code - Parameter Initialization

```python
def initialize_parameters_deep(layer_dims, initialization="he"):
    """
    Initialize parameters for L-layer neural network

    Arguments:
    layer_dims -- list containing dimensions of each layer
    initialization -- "he", "xavier", or "random"

    Returns:
    parameters -- dictionary containing W1, b1, ..., WL, bL
    """
    np.random.seed(42)
    parameters = {}
    L = len(layer_dims) - 1  # number of layers

    for l in range(1, L + 1):
        if initialization == "he":
            # He initialization: good for ReLU
            # Formula: W ~ N(0, sqrt(2/n^[l-1]))
            parameters[f'W{l}'] = np.random.randn(layer_dims[l], layer_dims[l-1]) * np.sqrt(2 / layer_dims[l-1])
        elif initialization == "xavier":
            # Xavier initialization: good for sigmoid/tanh
            # Formula: W ~ N(0, sqrt(1/n^[l-1]))
            parameters[f'W{l}'] = np.random.randn(layer_dims[l], layer_dims[l-1]) * np.sqrt(1 / layer_dims[l-1])
        else:
            # Random small initialization
            parameters[f'W{l}'] = np.random.randn(layer_dims[l], layer_dims[l-1]) * 0.01

        # Bias initialization (always zeros)
        parameters[f'b{l}'] = np.zeros((layer_dims[l], 1))

    return parameters
```

```python
def linear_forward(A, W, b):
    """
    Implement the linear part of forward propagation

    Mathematical Formula:
    Z = W·A + b

    Arguments:
    A -- activations from previous layer: (size of previous layer, number of examples)
    W -- weights matrix: (size of current layer, size of previous layer)
    b -- bias vector: (size of current layer, 1)

    Returns:
    Z -- the input of the activation function
    cache -- tuple containing A, W, b for backward propagation
    """
    # Forward propagation calculation
    Z = np.dot(W, A) + b

    # Store values for backward propagation
    cache = (A, W, b)

    # Dimension check
    assert(Z.shape == (W.shape[0], A.shape[1]))

    return Z, cache

def linear_activation_forward(A_prev, W, b, activation):
    """
    Implement forward propagation for LINEAR->ACTIVATION layer

    Arguments:
    A_prev -- activations from previous layer
    W -- weights matrix
    b -- bias vector
    activation -- activation function: "sigmoid", "relu", "tanh"

    Returns:
    A -- output of activation function
    cache -- tuple containing linear_cache and activation_cache
    """
    # Linear forward
    Z, linear_cache = linear_forward(A_prev, W, b)

    # Apply activation function
    if activation == "sigmoid":
        A = sigmoid(Z)
    elif activation == "relu":
        A = relu(Z)
    elif activation == "tanh":
        A = tanh_activation(Z)
    else:
        raise ValueError(f"Unknown activation: {activation}")

    # Store caches
    activation_cache = Z
    cache = (linear_cache, activation_cache)

    return A, cache
```

```python
def L_model_forward(X, parameters, activations):
    """
    Implement forward propagation for L-layer neural network

    Mathematical Process:
    For each layer l = 1, 2, ..., L:
        Z^[l] = W^[l] × A^[l-1] + b^[l]
        A^[l] = g^[l](Z^[l])

    Arguments:
    X -- input data: (input size, number of examples)
    parameters -- output of initialize_parameters_deep()
    activations -- list of activation functions for each layer

    Returns:
    AL -- last post-activation value (predictions)
    caches -- list of caches for each layer
    """
    caches = []
    A = X
    L = len(parameters) // 2  # number of layers

    print(f"Forward propagation through {L} layers:")
    print(f"Input shape: {A.shape}")

    # Implement [LINEAR -> ACTIVATION] * (L-1)
    for l in range(1, L):
        A_prev = A
        A, cache = linear_activation_forward(A_prev,
                                           parameters[f'W{l}'],
                                           parameters[f'b{l}'],
                                           activations[l-1])
        caches.append(cache)
        print(f"Layer {l} ({activations[l-1]}): {A_prev.shape} -> {A.shape}")

    # Implement LINEAR -> SIGMOID for output layer
    AL, cache = linear_activation_forward(A,
                                        parameters[f'W{L}'],
                                        parameters[f'b{L}'],
                                        activations[L-1])
    caches.append(cache)
    print(f"Layer {L} ({activations[L-1]}): {A.shape} -> {AL.shape}")

    # Final check
    assert(AL.shape == (1, X.shape[1]))

    return AL, caches
```

## Cell 10: Markdown - Backward Propagation Theory

```markdown
## Backward Propagation in Deep Networks

Backward propagation computes gradients using the **chain rule** to enable parameter updates.

### Mathematical Foundation

For each layer l, we need to compute:

$$\frac{\partial \mathcal{L}}{\partial W^{[l]}} = \frac{\partial \mathcal{L}}{\partial Z^{[l]}} \frac{\partial Z^{[l]}}{\partial W^{[l]}}$$

$$\frac{\partial \mathcal{L}}{\partial b^{[l]}} = \frac{\partial \mathcal{L}}{\partial Z^{[l]}} \frac{\partial Z^{[l]}}{\partial b^{[l]}}$$

### Step-by-Step Process

1. **Start from output**: $dA^{[L]} = \frac{\partial \mathcal{L}}{\partial A^{[L]}}$

2. **For each layer l (from L to 1)**:
   - $dZ^{[l]} = dA^{[l]} \odot g'^{[l]}(Z^{[l]})$
   - $dW^{[l]} = \frac{1}{m} dZ^{[l]} (A^{[l-1]})^T$
   - $db^{[l]} = \frac{1}{m} \sum_{i=1}^{m} dZ^{[l](i)}$
   - $dA^{[l-1]} = (W^{[l]})^T dZ^{[l]}$

### Key Insights

- **Gradients flow backward** through the network
- **Chain rule** connects gradients across layers
- **Matrix dimensions** must match for correct computation
```

## Cell 11: Code - Linear Backward Propagation

```python
def linear_backward(dZ, cache):
    """
    Implement the linear portion of backward propagation for a single layer

    Mathematical Formulas:
    dW^[l] = (1/m) * dZ^[l] × (A^[l-1])^T
    db^[l] = (1/m) * sum(dZ^[l]) along axis 1
    dA^[l-1] = (W^[l])^T × dZ^[l]

    Arguments:
    dZ -- gradient of cost with respect to linear output of current layer
    cache -- tuple of values (A_prev, W, b) from forward propagation

    Returns:
    dA_prev -- gradient of cost w.r.t. activation of previous layer
    dW -- gradient of cost w.r.t. W (current layer)
    db -- gradient of cost w.r.t. b (current layer)
    """
    A_prev, W, b = cache
    m = A_prev.shape[1]

    # Compute gradients
    dW = (1 / m) * np.dot(dZ, A_prev.T)
    db = (1 / m) * np.sum(dZ, axis=1, keepdims=True)
    dA_prev = np.dot(W.T, dZ)

    # Dimension checks
    assert(dA_prev.shape == A_prev.shape)
    assert(dW.shape == W.shape)
    assert(db.shape == b.shape)

    return dA_prev, dW, db

def linear_activation_backward(dA, cache, activation):
    """
    Implement backward propagation for LINEAR->ACTIVATION layer

    Arguments:
    dA -- post-activation gradient for current layer
    cache -- tuple of values (linear_cache, activation_cache)
    activation -- activation function used: "relu", "sigmoid", "tanh"

    Returns:
    dA_prev -- gradient of cost w.r.t. activation of previous layer
    dW -- gradient of cost w.r.t. W (current layer)
    db -- gradient of cost w.r.t. b (current layer)
    """
    linear_cache, activation_cache = cache

    # Compute dZ based on activation function
    if activation == "relu":
        dZ = relu_backward(dA, activation_cache)
    elif activation == "sigmoid":
        dZ = sigmoid_backward(dA, activation_cache)
    elif activation == "tanh":
        dZ = tanh_backward(dA, activation_cache)
    else:
        raise ValueError(f"Unknown activation: {activation}")

    # Compute linear backward
    dA_prev, dW, db = linear_backward(dZ, linear_cache)

    return dA_prev, dW, db
```

## Cell 12: Code - Complete Backward Propagation

```python
def L_model_backward(AL, Y, caches, activations):
    """
    Implement backward propagation for L-layer neural network

    Arguments:
    AL -- probability vector, output of forward propagation (L_model_forward())
    Y -- true "label" vector
    caches -- list of caches from L_model_forward()
    activations -- list of activation functions used

    Returns:
    grads -- dictionary with gradients
    """
    grads = {}
    L = len(caches)  # number of layers
    m = AL.shape[1]
    Y = Y.reshape(AL.shape)  # Ensure Y has same shape as AL

    print(f"Backward propagation through {L} layers:")
    print(f"Output AL shape: {AL.shape}, Y shape: {Y.shape}")

    # Initialize backpropagation
    # For binary cross-entropy with sigmoid: dAL = -(Y/AL - (1-Y)/(1-AL))
    # But for sigmoid + cross-entropy, we can simplify to: dZ = AL - Y

    # Output layer: LINEAR -> SIGMOID
    current_cache = caches[L-1]
    grads[f"dA{L}"] = -(np.divide(Y, AL) - np.divide(1 - Y, 1 - AL))
    grads[f"dA{L-1}"], grads[f"dW{L}"], grads[f"db{L}"] = \
        linear_activation_backward(grads[f"dA{L}"], current_cache, activations[L-1])

    print(f"Layer {L}: Computed dW{L} {grads[f'dW{L}'].shape}, db{L} {grads[f'db{L}'].shape}")

    # Hidden layers: LINEAR -> ACTIVATION (ReLU/tanh)
    for l in reversed(range(L-1)):
        current_cache = caches[l]
        dA_prev_temp, dW_temp, db_temp = \
            linear_activation_backward(grads[f"dA{l+2}"], current_cache, activations[l])

        grads[f"dA{l+1}"] = dA_prev_temp
        grads[f"dW{l+1}"] = dW_temp
        grads[f"db{l+1}"] = db_temp

        print(f"Layer {l+1}: Computed dW{l+1} {dW_temp.shape}, db{l+1} {db_temp.shape}")

    return grads

def compute_cost(AL, Y):
    """
    Implement the cost function for binary classification

    Formula: J = -(1/m) * Σ[Y*log(AL) + (1-Y)*log(1-AL)]

    Arguments:
    AL -- probability vector of shape (1, number of examples)
    Y -- true "label" vector of shape (1, number of examples)

    Returns:
    cost -- cross-entropy cost
    """
    m = Y.shape[1]

    # Clip AL to prevent log(0)
    AL_clipped = np.clip(AL, 1e-8, 1 - 1e-8)

    # Compute cost
```
