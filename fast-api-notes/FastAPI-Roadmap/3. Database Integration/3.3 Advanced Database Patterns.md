# 3.3 Advanced Database Patterns in FastAPI

## Overview

Advanced database patterns help create maintainable, scalable, and testable FastAPI applications. These patterns provide separation of concerns, improve code reusability, and enable better testing strategies while maintaining high performance.

## Repository Pattern

### Concept and Benefits

The Repository Pattern encapsulates database access logic and provides a clean separation between business logic and data access layers.

**Benefits:**

- **Testability**: Easy to mock for unit testing
- **Maintainability**: Centralized data access logic
- **Flexibility**: Switch between different data sources
- **Consistency**: Standardized data operations

### Basic Repository Implementation

```python
# repositories/base.py
from abc import ABC, abstractmethod
from typing import Generic, TypeVar, List, Optional
from sqlalchemy.orm import Session

T = TypeVar('T')

class BaseRepository(ABC, Generic[T]):
    @abstractmethod
    async def create(self, entity: T) -> T:
        pass

    @abstractmethod
    async def get_by_id(self, id: int) -> Optional[T]:
        pass

    @abstractmethod
    async def get_all(self, skip: int = 0, limit: int = 100) -> List[T]:
        pass

    @abstractmethod
    async def update(self, id: int, entity: T) -> Optional[T]:
        pass

    @abstractmethod
    async def delete(self, id: int) -> bool:
        pass
```

### SQL Repository Implementation

```python
# repositories/user_repository.py
from sqlalchemy.orm import Session
from sqlalchemy import and_, or_
from models import User
from schemas import UserCreate, UserUpdate
from repositories.base import BaseRepository
from typing import List, Optional

class UserRepository(BaseRepository[User]):
    def __init__(self, db: Session):
        self.db = db

    async def create(self, user_data: UserCreate) -> User:
        user = User(**user_data.dict())
        self.db.add(user)
        self.db.commit()
        self.db.refresh(user)
        return user

    async def get_by_id(self, user_id: int) -> Optional[User]:
        return self.db.query(User).filter(User.id == user_id).first()

    async def get_by_email(self, email: str) -> Optional[User]:
        return self.db.query(User).filter(User.email == email).first()

    async def get_all(self, skip: int = 0, limit: int = 100) -> List[User]:
        return self.db.query(User).offset(skip).limit(limit).all()

    async def update(self, user_id: int, user_data: UserUpdate) -> Optional[User]:
        user = self.db.query(User).filter(User.id == user_id).first()
        if user:
            for field, value in user_data.dict(exclude_unset=True).items():
                setattr(user, field, value)
            self.db.commit()
            self.db.refresh(user)
        return user

    async def delete(self, user_id: int) -> bool:
        user = self.db.query(User).filter(User.id == user_id).first()
        if user:
            self.db.delete(user)
            self.db.commit()
            return True
        return False

    # Custom query methods
    async def search_users(self, query: str) -> List[User]:
        return self.db.query(User).filter(
            or_(
                User.name.contains(query),
                User.email.contains(query)
            )
        ).all()

    async def get_active_users(self) -> List[User]:
        return self.db.query(User).filter(User.is_active == True).all()
```

## Unit of Work Pattern

### Concept and Implementation

Unit of Work pattern maintains a list of objects affected by a business transaction and coordinates writing out changes and resolving concurrency problems.

```python
# uow/base.py
from abc import ABC, abstractmethod
from typing import Dict, Any

class AbstractUnitOfWork(ABC):
    def __init__(self):
        self.committed = False

    def __enter__(self):
        return self

    def __exit__(self, *args):
        if not self.committed:
            self.rollback()

    @abstractmethod
    def commit(self):
        pass

    @abstractmethod
    def rollback(self):
        pass

# uow/sqlalchemy_uow.py
from sqlalchemy.orm import Session
from sqlalchemy.orm import sessionmaker
from repositories.user_repository import UserRepository
from repositories.post_repository import PostRepository

class SqlAlchemyUnitOfWork(AbstractUnitOfWork):
    def __init__(self, session_factory: sessionmaker):
        super().__init__()
        self.session_factory = session_factory

    def __enter__(self):
        self.session: Session = self.session_factory()
        self.users = UserRepository(self.session)
        self.posts = PostRepository(self.session)
        return super().__enter__()

    def __exit__(self, *args):
        super().__exit__(*args)
        self.session.close()

    def commit(self):
        self.session.commit()
        self.committed = True

    def rollback(self):
        self.session.rollback()
```

### Service Layer with Unit of Work

```python
# services/user_service.py
from uow.sqlalchemy_uow import SqlAlchemyUnitOfWork
from schemas import UserCreate, PostCreate
from exceptions import EmailAlreadyExistsError, UserNotFoundError

class UserService:
    def __init__(self, uow: SqlAlchemyUnitOfWork):
        self.uow = uow

    async def create_user_with_post(self, user_data: UserCreate, post_data: PostCreate):
        with self.uow:
            # Check if email already exists
            existing_user = await self.uow.users.get_by_email(user_data.email)
            if existing_user:
                raise EmailAlreadyExistsError("Email already registered")

            # Create user
            user = await self.uow.users.create(user_data)

            # Create initial post
            post_data.author_id = user.id
            post = await self.uow.posts.create(post_data)

            # Commit transaction
            self.uow.commit()

            return {"user": user, "post": post}

    async def transfer_posts(self, from_user_id: int, to_user_id: int):
        with self.uow:
            # Verify both users exist
            from_user = await self.uow.users.get_by_id(from_user_id)
            to_user = await self.uow.users.get_by_id(to_user_id)

            if not from_user or not to_user:
                raise UserNotFoundError("One or both users not found")

            # Transfer all posts
            posts = await self.uow.posts.get_by_author_id(from_user_id)
            for post in posts:
                post.author_id = to_user_id
                await self.uow.posts.update(post.id, post)

            self.uow.commit()

            return {"transferred_posts": len(posts)}
```

```python
# dao/user_dao.py
from typing import List, Optional, Dict, Any
from sqlalchemy.orm import Session
from sqlalchemy import text
from models import User

class UserDAO:
    def __init__(self, db: Session):
        self.db = db

    async def execute_raw_query(self, query: str, params: Dict[str, Any] = None) -> List[Dict]:
        result = self.db.execute(text(query), params or {})
        return [dict(row) for row in result.fetchall()]

    async def get_user_statistics(self) -> Dict[str, Any]:
        query = """
        SELECT
            COUNT(*) as total_users,
            COUNT(CASE WHEN is_active = true THEN 1 END) as active_users,
            COUNT(CASE WHEN created_at > NOW() - INTERVAL '30 days' THEN 1 END) as recent_users
        FROM users
        """
        result = await self.execute_raw_query(query)
        return result[0] if result else {}

    async def get_users_with_post_count(self, min_posts: int = 0) -> List[Dict]:
        query = """
        SELECT u.id, u.name, u.email, COUNT(p.id) as post_count
        FROM users u
        LEFT JOIN posts p ON u.id = p.author_id
        GROUP BY u.id, u.name, u.email
        HAVING COUNT(p.id) >= :min_posts
        ORDER BY post_count DESC
        """
        return await self.execute_raw_query(query, {"min_posts": min_posts})

    async def bulk_update_user_status(self, user_ids: List[int], status: str) -> int:
        query = """
        UPDATE users
        SET is_active = :status, updated_at = NOW()
        WHERE id = ANY(:user_ids)
        """
        result = self.db.execute(text(query), {"status": status, "user_ids": user_ids})
        self.db.commit()
        return result.rowcount
```

## Query Optimization Patterns

### Query Builder Pattern

```python
# query_builders/user_query_builder.py
from sqlalchemy.orm import Query
from sqlalchemy import and_, or_, desc, asc
from models import User
from typing import Optional, List
from datetime import datetime

class UserQueryBuilder:
    def __init__(self, base_query: Query):
        self.query = base_query

    def filter_by_status(self, is_active: bool):
        self.query = self.query.filter(User.is_active == is_active)
        return self

    def filter_by_email_domain(self, domain: str):
        self.query = self.query.filter(User.email.like(f"%@{domain}"))
        return self

    def filter_by_date_range(self, start_date: datetime, end_date: datetime):
        self.query = self.query.filter(
            and_(
                User.created_at >= start_date,
                User.created_at <= end_date
            )
        )
        return self

    def search_by_name_or_email(self, search_term: str):
        self.query = self.query.filter(
            or_(
                User.name.contains(search_term),
                User.email.contains(search_term)
            )
        )
        return self

    def order_by_created_date(self, ascending: bool = True):
        order_func = asc if ascending else desc
        self.query = self.query.order_by(order_func(User.created_at))
        return self

    def paginate(self, page: int, page_size: int):
        offset = (page - 1) * page_size
        self.query = self.query.offset(offset).limit(page_size)
        return self

    def build(self) -> Query:
        return self.query

# Usage
def get_filtered_users(db: Session, filters: Dict[str, Any]) -> List[User]:
    builder = UserQueryBuilder(db.query(User))

    if filters.get("is_active") is not None:
        builder.filter_by_status(filters["is_active"])

    if filters.get("email_domain"):
        builder.filter_by_email_domain(filters["email_domain"])

    if filters.get("search_term"):
        builder.search_by_name_or_email(filters["search_term"])

    if filters.get("start_date") and filters.get("end_date"):
        builder.filter_by_date_range(filters["start_date"], filters["end_date"])

    builder.order_by_created_date(filters.get("ascending", True))

    if filters.get("page") and filters.get("page_size"):
        builder.paginate(filters["page"], filters["page_size"])

    return builder.build().all()
```

### Advanced Connection Configuration

```python
from sqlalchemy import create_engine, event
from sqlalchemy.pool import QueuePool
from sqlalchemy.orm import sessionmaker
import logging

class DatabaseManager:
    def __init__(self, database_url: str, **kwargs):

            pool_size=20,              # Number of connections to maintain
            max_overflow=30,           # Additional connections beyond pool_size
            pool_pre_ping=True,        # Validate connections before use
```

### Read/Write Split Pattern

```python
# database/read_write_split.py
from enum import Enum
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
from contextlib import contextmanager

class DatabaseType(Enum):
    READ = "read"
    WRITE = "write"

class ReadWriteDBManager:
    def __init__(self, read_url: str, write_url: str):
        self.read_engine = create_engine(read_url, pool_pre_ping=True)
        self.write_engine = create_engine(write_url, pool_pre_ping=True)

        self.ReadSession = sessionmaker(bind=self.read_engine)
        self.WriteSession = sessionmaker(bind=self.write_engine)

    @contextmanager
    def get_session(self, db_type: DatabaseType):
        if db_type == DatabaseType.READ:
            session = self.ReadSession()
        else:
            session = self.WriteSession()

        try:
            yield session
        finally:
            session.close()

    def get_read_session(self):
        return self.ReadSession()

    def get_write_session(self):
        return self.WriteSession()

# Usage in repository
class OptimizedUserRepository:
    def __init__(self, db_manager: ReadWriteDBManager):
        self.db_manager = db_manager

    async def get_user(self, user_id: int):
        # Use read replica for queries
        with self.db_manager.get_session(DatabaseType.READ) as session:
            return session.query(User).filter(User.id == user_id).first()

    async def create_user(self, user_data: UserCreate):
        # Use write master for modifications
        with self.db_manager.get_session(DatabaseType.WRITE) as session:
            user = User(**user_data.dict())
            session.add(user)
            session.commit()
            session.refresh(user)
            return user
```

## Caching Patterns

### Multi-Level Caching

```python
# caching/cache_manager.py
from typing import Any, Optional, Dict
import json
import pickle
from functools import wraps
import redis.asyncio as redis
from sqlalchemy.orm import Session

class CacheManager:
    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client
        self.local_cache: Dict[str, Any] = {}
        self.max_local_cache_size = 1000

    async def get(self, key: str) -> Optional[Any]:
        # Try local cache first (L1)
        if key in self.local_cache:
            return self.local_cache[key]

        # Try Redis cache (L2)
        value = await self.redis.get(key)
        if value:
            try:
                deserialized = pickle.loads(value)
                # Store in local cache
                self._set_local_cache(key, deserialized)
                return deserialized
            except Exception:
                pass

        return None

    async def set(self, key: str, value: Any, expire: int = None):
        # Set in Redis
        serialized = pickle.dumps(value)
        await self.redis.set(key, serialized, ex=expire)

        # Set in local cache
        self._set_local_cache(key, value)

    def _set_local_cache(self, key: str, value: Any):
        if len(self.local_cache) >= self.max_local_cache_size:
            # Remove oldest entry (simple LRU)
            oldest_key = next(iter(self.local_cache))
            del self.local_cache[oldest_key]

        self.local_cache[key] = value

# Cache decorator
def cached(expire: int = 300, key_prefix: str = ""):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            cache_key = f"{key_prefix}{func.__name__}:{hash(str(args) + str(kwargs))}"

            # Try cache first
            cached_result = await cache_manager.get(cache_key)
            if cached_result is not None:
                return cached_result

            # Execute function
            result = await func(*args, **kwargs)

            # Cache result
            await cache_manager.set(cache_key, result, expire=expire)

            return result
        return wrapper
    return decorator
```

```python
# patterns/cache_aside.py
class CacheAsideUserRepository:
    def __init__(self, db: Session, cache: CacheManager):
        self.db = db
        self.cache = cache

    async def get_user(self, user_id: int) -> Optional[User]:
        cache_key = f"user:{user_id}"

            return cached_user

        # Cache the result
        if user:
```

## Database Sharding Patterns

### Horizontal Sharding

```python
# sharding/shard_manager.py
from typing import Dict, List
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
import hashlib

class ShardManager:
    def __init__(self, shard_configs: Dict[str, str]):
        self.shards = {}
        self.sessions = {}

        for shard_name, database_url in shard_configs.items():
            engine = create_engine(database_url)
            self.shards[shard_name] = engine
            self.sessions[shard_name] = sessionmaker(bind=engine)

    def get_shard_for_user(self, user_id: int) -> str:
        """Determine shard based on user ID"""
        shard_hash = int(hashlib.md5(str(user_id).encode()).hexdigest(), 16)
        shard_index = shard_hash % len(self.shards)
        return list(self.shards.keys())[shard_index]

    def get_session_for_user(self, user_id: int):
        shard_name = self.get_shard_for_user(user_id)
        return self.sessions[shard_name]()

    def get_all_sessions(self):
        return [session_class() for session_class in self.sessions.values()]

class ShardedUserRepository:
    def __init__(self, shard_manager: ShardManager):
        self.shard_manager = shard_manager

    async def get_user(self, user_id: int) -> Optional[User]:
        session = self.shard_manager.get_session_for_user(user_id)
        try:
            return session.query(User).filter(User.id == user_id).first()
        finally:
            session.close()

    async def search_all_shards(self, search_term: str) -> List[User]:
        results = []
        sessions = self.shard_manager.get_all_sessions()

        try:
            for session in sessions:
                shard_results = session.query(User).filter(
                    User.name.contains(search_term)
                ).all()
                results.extend(shard_results)
        finally:
            for session in sessions:
                session.close()

        return results
```

```python
        self.slow_query_threshold = 1.0  # seconds
        self.setup_monitoring()

    def setup_monitoring(self):
        @event.listens_for(Engine, "before_cursor_execute")
        def receive_before_cursor_execute(conn, cursor, statement, parameters, context, executemany):
            context._query_start_time = time.time()

        @event.listens_for(Engine, "after_cursor_execute")
        def receive_after_cursor_execute(conn, cursor, statement, parameters, context, executemany):
            total_time = time.time() - context._query_start_time

            if total_time > self.slow_query_threshold:
                logging.warning(
                    f"Slow query detected: {total_time:.2f}s\n"
                    f"Statement: {statement}\n"
                    f"Parameters: {parameters}"
                )
```

```python
# cqrs/handlers.py
from abc import ABC, abstractmethod
from typing import Any, Optional

class CommandHandler(ABC):
    @abstractmethod
    async def handle(self, command: Command) -> Any:
        pass

class QueryHandler(ABC):
    @abstractmethod
    async def handle(self, query: Query) -> Any:
        pass

class CreateUserCommandHandler(CommandHandler):
    def __init__(self, write_db: Session, event_store: EventStore):
        self.write_db = write_db
        self.event_store = event_store

    async def handle(self, command: CreateUserCommand) -> User:
        # Create user in write database
        user = User(name=command.name, email=command.email, age=command.age)
        self.write_db.add(user)
        self.write_db.commit()
        self.write_db.refresh(user)

        # Store event for read model updates
        event = Event(
            aggregate_id=str(user.id),
            event_type="UserCreated",
            event_data=command.dict(),
            timestamp=command.timestamp,
            version=1
        )
        await self.event_store.append_event(event)

        return user

class GetUserQueryHandler(QueryHandler):
    def __init__(self, read_db: Session, cache: CacheManager):
        self.read_db = read_db
        self.cache = cache

    async def handle(self, query: GetUserQuery) -> Optional[User]:
        # Try cache first
        cache_key = f"user:{query.user_id}"
        cached_user = await self.cache.get(cache_key)
        if cached_user:
            return cached_user

        # Query read database (potentially optimized read model)
        user = self.read_db.query(UserReadModel).filter(
            UserReadModel.id == query.user_id
        ).first()

        if user:
            await self.cache.set(cache_key, user, expire=300)

        return user
```

```python
from typing import Dict, Type, Any

class CommandBus:
```

## Saga Pattern for Distributed Transactions

### Saga Implementation

```python
# saga/saga_manager.py
from enum import Enum
from typing import List, Dict, Any, Optional
from datetime import datetime
import uuid

class SagaStatus(Enum):
    STARTED = "started"
    COMPLETED = "completed"
    FAILED = "failed"
    COMPENSATING = "compensating"

class SagaStep:
    def __init__(self, name: str, action, compensation):
        self.name = name
        self.action = action
        self.compensation = compensation
        self.executed = False
        self.compensated = False

class Saga:
    def __init__(self, saga_id: str = None):
        self.saga_id = saga_id or str(uuid.uuid4())
        self.steps: List[SagaStep] = []
        self.status = SagaStatus.STARTED
        self.current_step = 0
        self.created_at = datetime.utcnow()

    def add_step(self, name: str, action, compensation):
        step = SagaStep(name, action, compensation)
        self.steps.append(step)

    async def execute(self):
        try:
            for i, step in enumerate(self.steps):
                self.current_step = i
                await step.action()
                step.executed = True

            self.status = SagaStatus.COMPLETED
        except Exception as e:
            self.status = SagaStatus.FAILED
            await self._compensate()
            raise

    async def _compensate(self):
        self.status = SagaStatus.COMPENSATING

        # Execute compensations in reverse order
        for step in reversed(self.steps[:self.current_step + 1]):
            if step.executed and not step.compensated:
                try:
                    await step.compensation()
                    step.compensated = True
                except Exception as e:
                    logging.error(f"Compensation failed for step {step.name}: {e}")

# Example: User Registration Saga
class UserRegistrationSaga:
    def __init__(self, user_service, email_service, notification_service):
        self.user_service = user_service
        self.email_service = email_service
        self.notification_service = notification_service

    async def register_user(self, user_data: UserCreate):
        saga = Saga()

        # Step 1: Create user account
        saga.add_step(
            "create_user",
            lambda: self.user_service.create_user(user_data),
            lambda: self.user_service.delete_user(user_data.email)
        )

        # Step 2: Send welcome email
        saga.add_step(
            "send_welcome_email",
            lambda: self.email_service.send_welcome_email(user_data.email),
            lambda: self.email_service.send_cancellation_email(user_data.email)
        )

        # Step 3: Create user profile
        saga.add_step(
            "create_profile",
            lambda: self.user_service.create_profile(user_data),
            lambda: self.user_service.delete_profile(user_data.email)
        )

        await saga.execute()
        return saga.saga_id
```

## Database Migration Patterns

### Advanced Migration Strategies

```python
# migrations/migration_manager.py
from typing import List, Callable
from abc import ABC, abstractmethod
import logging

class Migration(ABC):
    def __init__(self, version: str, description: str):
        self.version = version
        self.description = description

    @abstractmethod
    async def up(self, db: Session):
        pass

    @abstractmethod
    async def down(self, db: Session):
        pass

class DataMigration(Migration):
    """For migrating data without schema changes"""
    def __init__(self, version: str, description: str, migration_func: Callable):
        super().__init__(version, description)
        self.migration_func = migration_func

    async def up(self, db: Session):
        await self.migration_func(db)

    async def down(self, db: Session):
        # Data migrations may not be reversible
        logging.warning(f"Data migration {self.version} is not reversible")

class SchemaMigration(Migration):
    """For schema changes with proper rollback"""
    def __init__(self, version: str, description: str, up_sql: str, down_sql: str):
        super().__init__(version, description)
        self.up_sql = up_sql
        self.down_sql = down_sql

    async def up(self, db: Session):
        db.execute(text(self.up_sql))
        db.commit()

    async def down(self, db: Session):
        db.execute(text(self.down_sql))
        db.commit()

class MigrationManager:
    def __init__(self, db: Session):
        self.db = db
        self.migrations: List[Migration] = []

    def add_migration(self, migration: Migration):
        self.migrations.append(migration)

    async def migrate_up(self, target_version: str = None):
        applied_migrations = self._get_applied_migrations()

        for migration in self.migrations:
            if migration.version not in applied_migrations:
                logging.info(f"Applying migration {migration.version}: {migration.description}")
                await migration.up(self.db)
                self._record_migration(migration.version)

                if target_version and migration.version == target_version:
                    break

    async def migrate_down(self, target_version: str):
        applied_migrations = self._get_applied_migrations()

        for migration in reversed(self.migrations):
            if migration.version in applied_migrations:
                if migration.version == target_version:
                    break

                logging.info(f"Rolling back migration {migration.version}")
                await migration.down(self.db)
                self._remove_migration_record(migration.version)
```

## Audit Trail Pattern

### Audit Implementation

```python
# audit/audit_manager.py
from sqlalchemy import Column, Integer, String, DateTime, Text, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from datetime import datetime
from typing import Any, Dict
import json

class AuditLog(Base):
    __tablename__ = "audit_logs"

    id = Column(Integer, primary_key=True)
    table_name = Column(String, nullable=False)
    record_id = Column(String, nullable=False)
    action = Column(String, nullable=False)  # INSERT, UPDATE, DELETE
    old_values = Column(Text)  # JSON
    new_values = Column(Text)  # JSON
    user_id = Column(Integer, ForeignKey("users.id"))
    timestamp = Column(DateTime, default=datetime.utcnow)
    ip_address = Column(String)

class AuditManager:
    def __init__(self, db: Session):
        self.db = db

    async def log_change(self, table_name: str, record_id: Any, action: str,
                        old_values: Dict = None, new_values: Dict = None,
                        user_id: int = None, ip_address: str = None):
        audit_log = AuditLog(
            table_name=table_name,
            record_id=str(record_id),
            action=action,
            old_values=json.dumps(old_values) if old_values else None,
            new_values=json.dumps(new_values) if new_values else None,
            user_id=user_id,
            ip_address=ip_address
        )

        self.db.add(audit_log)
        self.db.commit()

    async def get_audit_trail(self, table_name: str, record_id: Any) -> List[AuditLog]:
        return self.db.query(AuditLog).filter(
            AuditLog.table_name == table_name,
            AuditLog.record_id == str(record_id)
        ).order_by(AuditLog.timestamp.desc()).all()

# Audit decorator for repository methods
def audit_changes(table_name: str):
    def decorator(func):
        @wraps(func)
        async def wrapper(self, *args, **kwargs):
            # Get user context from request
            user_id = getattr(self, 'current_user_id', None)
            ip_address = getattr(self, 'current_ip', None)

            # Execute original function
            result = await func(self, *args, **kwargs)

            # Log the change
            if hasattr(self, 'audit_manager'):
                action = func.__name__.upper()
                record_id = getattr(result, 'id', None) if result else None

                if record_id:
                    await self.audit_manager.log_change(
                        table_name=table_name,
                        record_id=record_id,
                        action=action,
                        new_values=result.__dict__ if result else None,
                        user_id=user_id,
                        ip_address=ip_address
                    )

            return result
        return wrapper
    return decorator
```

## Soft Delete Pattern

### Soft Delete Implementation

```python
# patterns/soft_delete.py
from sqlalchemy import Column, Boolean, DateTime
from datetime import datetime

class SoftDeleteMixin:
    is_deleted = Column(Boolean, default=False, nullable=False)
    deleted_at = Column(DateTime, nullable=True)
    deleted_by = Column(Integer, ForeignKey("users.id"), nullable=True)

class SoftDeleteManager:
    @staticmethod
    def soft_delete(db: Session, model_class, record_id: int, user_id: int = None):
        record = db.query(model_class).filter(model_class.id == record_id).first()
        if record:
            record.is_deleted = True
            record.deleted_at = datetime.utcnow()
            record.deleted_by = user_id
            db.commit()
            return record
        return None

    @staticmethod
    def restore(db: Session, model_class, record_id: int):
        record = db.query(model_class).filter(
            model_class.id == record_id,
            model_class.is_deleted == True
        ).first()
        if record:
            record.is_deleted = False
            record.deleted_at = None
            record.deleted_by = None
            db.commit()
            return record
        return None

    @staticmethod
    def get_active_records(db: Session, model_class):
        return db.query(model_class).filter(model_class.is_deleted == False)

    @staticmethod
    def get_deleted_records(db: Session, model_class):
        return db.query(model_class).filter(model_class.is_deleted == True)

# Enhanced repository with soft delete
class SoftDeleteRepository(BaseRepository[T]):
    def __init__(self, db: Session, model_class):
        self.db = db
        self.model_class = model_class

    async def get_all(self, include_deleted: bool = False, skip: int = 0, limit: int = 100):
        query = self.db.query(self.model_class)

        if not include_deleted:
            query = query.filter(self.model_class.is_deleted == False)

        return query.offset(skip).limit(limit).all()

    async def delete(self, record_id: int, user_id: int = None) -> bool:
        return SoftDeleteManager.soft_delete(
            self.db, self.model_class, record_id, user_id
        ) is not None

    async def restore(self, record_id: int) -> bool:
        return SoftDeleteManager.restore(
            self.db, self.model_class, record_id
        ) is not None
```

```python
from sqlalchemy import create_engine, text
from sqlalchemy.orm import sessionmaker
from typing import Dict, Optional

class SchemaTenantManager:
    def __init__(self, base_database_url: str):

        self.tenant_engines: Dict[str, Engine] = {}
        self.tenant_sessions: Dict[str, sessionmaker] = {}

        with engine.connect() as conn:
            conn.execute(text(f"CREATE SCHEMA IF NOT EXISTS tenant_{tenant_id}"))
            conn.commit()

            engine = create_engine(tenant_url)
            self.tenant_engines[tenant_id] = engine
            self.tenant_sessions[tenant_id] = sessionmaker(bind=engine)

        return self.tenant_sessions[tenant_id]
```
